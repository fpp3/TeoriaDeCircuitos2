\begin{circuitikz}[american, scale=0.80, transform shape]

    \draw (0, 0.5) node[left] {Lazo $(+)$} to[short, o-, f=$I_{\text{in}}$] (1.2, 0.5) coordinate(p_in);

    \draw (rcm_top) to[R, l_={$R_{\text{CM}} = 1\,\mathrm{M}\Omega$}, *-] (2.2, 1.8) node[ground] {};
    
    % Buffer U1A (asociado a Lazo -)

    \draw (4.5, 3.5) node[op amp] (u1a) {};
    \node[font=\footnotesize] at ($(u1a.center)+(0.1,0)$) {$U_{1A}$};
    \draw (u1a_in) -- (u1a.+);
    \draw (u1a.-) -- ++(-0.2, 0) -- ++(0, 0.7) -| ($(u1a.out)+(0.1, 0)$);
    
    % Buffer U1B (asociado a Lazo +)
    \draw (p_in) to[short] (3.0, 0.5) -- (3.0, 1.0) coordinate(u1b_in);
    \draw (4.5, 0.5) node[op amp, noinv input up] (u1b) {};

    \draw (u1b.-) -- ++(-0.2, 0) -- ++(0, -0.7) -| ($(u1b.out)+(0.1, 0)$);

    % Conexion U1A -> R3 -> U1C(-)
    \draw ($(u1a.out)+(0.1, 0)$) -- (6.0, 3.5) -- (6.0, 2.5) to[R, l={$R_3$}] (8.1, 2.5) coordinate(n_minus) -- (u1c.-);

    % Conexion U1B -> R4 -> U1C(+)
    \draw ($(u1b.out)+(0.1, 0)$) -- (6.0, 0.5) -- (6.0, 1.5) to[R, l_={$R_4$}] (8.1, 1.5) coordinate(n_plus) -- (u1c.+);
    \draw (n_plus) to[R, l={$R_5$}, *-] (8.1, -0.2) node[ground] {};
    
    % Salida VE1
    \draw (11.0, 2.0) to[short, *-o] (12.2, 2.0) node[right] {$V_{E1}\,(1\text{--}5\,\mathrm{V})$};
