\begin{tikzpicture}
    \pgfplotsset{
        nice plot/.style={
            /pgf/number format/read comma as period,
            grid=both,
            major grid style={solid, gray!50},
            minor grid style={dotted, gray!50},
            axis line style={black},
            axis x line* = bottom,
            axis y line* = left,
            extra description/.code={
                \draw[gray!50] (rel axis cs:0,1) -- (rel axis cs:1,1) -- (rel axis cs:1,0);
            },
            width=15cm,
            height=7cm,
            xmin=0.1, xmax=1000,
            minor x tick num=9,
            minor y tick num=0,
        }
    }

    \begin{semilogxaxis}[
        nice plot,
        ylabel={Magnitud [dB]},
        ymin=-40, ymax=30,
        ytick={-40, -30, -20, -10, 0, 10, 20, 30},
    ]
        % Graficamos la magnitud ideal en dB
        \addplot[thick, blue, domain=0.1:1000, samples=200] {
            20*log10(3200 * 2 * pi * x) - 10*log10((2 * pi * x)^2 + 1600) - 10*log10((2 * pi * x)^2 + 25600)
        };
    \end{semilogxaxis}

    \begin{semilogxaxis}[
        nice plot,
        xlabel={$f$ [Hz]},
        ylabel={Fase [º]},
        ymin=-280, ymax=-80,
        ytick={-270, -225, -180, -135, -90},
        at={(0,-9.0cm)},
    ]
        % Graficamos la fase ideal en grados
        \addplot[thick, red, domain=0.1:1000, samples=200] {
            -90 - atan(2 * pi * x / 40) - atan(2 * pi * x / 160)
        };
    \end{semilogxaxis}
\end{tikzpicture}
